    3. JPA في بنية متعددة الطبقات

    لدراسة API JPA، استخدمنا بنية الاختبار التالية:

    كانت برامج الاختبار الخاصة بنا عبارة عن تطبيقات وحدة التحكم التي تستعلم مباشرة عن الطبقة JPA. اكتشفنا في هذه المناسبة الطرق الرئيسية للطبقة JPA. كنا في بيئة تُعرف باسم "Java SE" (الإصدار القياسي). يعمل JPA في كل من بيئة Java SE وبيئة Java EE5 (الإصدار المؤسسي).

    الآن بعد أن أصبحنا نتمتع بقدر من الإلمام بكل من تكوين الجسر العلائقي/الكائني واستخدام أساليب طبقة JPA، نعود إلى بنية متعددة الطبقات أكثر تقليدية:

    سيتم الوصول إلى الطبقة [JPA] عبر بنية ذات طبقتين [metier] و [dao]. سيتم استخدام إطار عمل Spring [7]، ثم الحاوية EJB3 من JBoss [8] لربط هذه الطبقات ببعضها البعض.

    ذكرنا أعلاه أن JPA متاح في بيئات SE و EE5. تقدم بيئة Java EE5 العديد من الخدمات في مجال الوصول إلى البيانات الدائمة، لا سيما مجموعات الاتصال ومديري المعاملات، ... وقد يكون من المفيد للمطور الاستفادة من هذه الخدمات. بيئة Java EE5 ليست منتشرة بعد (مايو 2007). وهي موجودة حاليًا على خادم التطبيقات Sun Application Server 9.x (Glassfish). خادم التطبيقات هو في الأساس خادم لتطبيقات الويب. إذا قمنا بإنشاء تطبيق رسومي مستقل من نوع Swing، فلن نتمكن من الاستفادة من بيئة EE والخدمات التي توفرها. وهذا يمثل مشكلة. بدأنا نرى بيئات "مستقلة" من نوع EE و c.a.d. يمكن استخدامها خارج خادم التطبيقات. هذا هو الحال بالنسبة لـ JBoss و EJB3 التي سنستخدمها في هذا المستند.

    في بيئة EE5، يتم تنفيذ الطبقات بواسطة كائنات تسمى EJB (Enterprise Java Bean). في الإصدارات السابقة من EE، يُعرف عن EJB (EJB 2.x) صعوبة تنفيذها واختبارها، وأحيانًا ضعف أدائها. يتم التمييز بين "entity" EJB2.x و"session" EJB2.x. باختصار، فإن "entity" EJB2.x هي صورة لصف في جدول قاعدة بيانات، و"session" EJB2.x هي كائن يُستخدم لتنفيذ طبقات [metier]، [dao] في بنية متعددة الطبقات. أحد الانتقادات الرئيسية الموجهة إلى الطبقات التي تم تنفيذها باستخدام EJB هو أنها لا يمكن استخدامها إلا داخل حاويات EJB، وهي خدمة تقدمها بيئة EE. وهذا يجعل الاختبارات الفردية مشكلة. وبالتالي، في المخطط أعلاه، ستتطلب الاختبارات الفردية للطبقات [metier] و [dao] التي تم إنشاؤها باستخدام EJB إعداد خادم تطبيقات، وهي عملية مرهقة إلى حد ما ولا تشجع المطور حقًا على إجراء الاختبارات بشكل متكرر.

    نشأ إطار عمل Spring كرد فعل على تعقيد بيئات EJB2. يوفر Spring في بيئة SE عددًا كبيرًا من الخدمات التي عادةً ما توفرها بيئات EE. وهكذا، في الجزء "استمرارية البيانات" الذي يهمنا هنا، يوفر Spring مجموعات الاتصال ومديري المعاملات التي تحتاجها التطبيقات. وقد ساهم ظهور Spring في تعزيز ثقافة الاختبارات الفردية، التي أصبحت فجأة أسهل بكثير في التنفيذ. يسمح Spring بتنفيذ طبقات التطبيق باستخدام كائنات Java التقليدية (POJO، Plain Old/Ordinary Java Object)، مما يتيح إعادة استخدامها في سياق آخر. وأخيرًا، يدمج العديد من أدوات الجهات الخارجية بطريقة شفافة إلى حد كبير، لا سيما أدوات الاستمرارية مثل Hibernate وIbatis، ...

    تم تصميم Java EE5 لتصحيح أوجه القصور في المواصفات السابقة EE. أصبحت EJB و 2.x هي EJB3. وهذه هي POJOs الموسومة بتعليقات توضيحية تجعلها كائنات خاصة عندما تكون داخل حاوية EJB3. في هذا الحاوية، سيتمكن EJB3 من الاستفادة من خدمات الحاوية (مجموعة الاتصالات، مدير المعاملات، ...). خارج الحاوية EJB3، يصبح EJB3 كائن Java عادي. يتم تجاهل تعليقاته التوضيحية EJB.

    فيما سبق، قمنا بتمثيل Spring و JBoss EJB3 كبنية تحتية (إطار عمل) محتملة لهندستنا متعددة الطبقات. هذه البنية التحتية هي التي ستوفر الخدمات التي نحتاجها: تجمع اتصالات ومدير معاملات.

    • مع Spring، سيتم تنفيذ الطبقات باستخدام POJOs. وستتمكن هذه من الوصول إلى خدمات Spring (مجموعة الاتصالات، مدير المعاملات) عن طريق حقن التبعيات في هذه POJOs: عند إنشاء هذه، يقوم Spring بحقنها بمراجع للخدمات التي ستحتاجها.
    • JBoss EJB3 هو حاوية EJB يمكنها العمل خارج خادم التطبيق. مبدأ عملها (بالنسبة للمطور) مشابه للمبدأ الموصوف لـ Spring. لن نجد سوى القليل من الاختلافات.

    التنفيذ [Dao] لهذه الواجهة هو كما يلي:

    
package dao;

import java.util.List;

import javax.persistence.EntityManager;
import javax.persistence.PersistenceContext;

import entites.Personne;

public class Dao implements IDao {

    @PersistenceContext
    private EntityManager em;

    // حذف شخص عبر معرّفه
    public void deleteOne(Integer id) {
        Personne personne = em.find(Personne.class, id);
        if (personne == null) {
            throw new DaoException(2);
        }
        em.remove(personne);
    }

    @SuppressWarnings("unchecked")
    // الحصول على جميع الأشخاص
    public List<Personne> getAll() {
        return em.createQuery("select p from Personne p").getResultList();
    }

    @SuppressWarnings("unchecked")
    // الحصول على الأشخاص الذين يتطابق اسمهم مع نموذج
    public List<Personne> getAllLike(String modele) {
        return em.createQuery("select p from Personne p where p.nom like :modele")
                .setParameter("modele", modele).getResultList();
    }

    // الحصول على شخص عبر معرّفه
    public Personne getOne(Integer id) {
        return em.find(Personne.class, id);
    }

    // حفظ شخص
    public Personne saveOne(Personne personne) {
        em.persist(personne);
        return personne;
    }

    // تحديث شخص
    public Personne updateOne(Personne personne) {
        return em.merge(personne);
    }

}

package dao;

@SuppressWarnings("serial")
public class DaoException extends RuntimeException {

    // رمز الخطأ
    private int code;

    public DaoException(int code) {
        super();
        this.code = code;
    }

    public DaoException(String message, int code) {
        super(message);
        this.code = code;
    }

    public DaoException(Throwable cause, int code) {
        super(cause);
        this.code = code;
    }

    public DaoException(String message, Throwable cause, int code) {
        super(message, cause);
        this.code = code;
    }

    // getter و setter

    public int getCode() {
        return code;
    }

    public void setCode(int code) {
        this.code = code;
    }

}

package dao;

@Stateless
public class Dao implements IDao {

    @PersistenceContext
    private EntityManager em;

    // حذف شخص عبر معرّفه
    @TransactionAttribute(TransactionAttributeType.REQUIRED)
    public void deleteOne(Integer id) {
        Personne personne = em.find(Personne.class, id);
        if (personne == null) {
            throw new DaoException(2);
        }
        em.remove(personne);
    }

    // الحصول على جميع الأشخاص
    @TransactionAttribute(TransactionAttributeType.REQUIRED)
    public List<Personne> getAll() {
        return em.createQuery("select p from Personne p").getResultList();
    }

    // الحصول على الأشخاص الذين يتطابق اسمهم مع نموذج
    @TransactionAttribute(TransactionAttributeType.REQUIRED)
    public List<Personne> getAllLike(String modele) {
        return em.createQuery("select p from Personne p where p.nom like :modele")
                .setParameter("modele", modele).getResultList();
    }

    // الحصول على شخص عبر معرّفه
    @TransactionAttribute(TransactionAttributeType.REQUIRED)
    public Personne getOne(Integer id) {
        return em.find(Personne.class, id);
    }

    // حفظ شخص
    @TransactionAttribute(TransactionAttributeType.REQUIRED)
    public Personne saveOne(Personne personne) {
        em.persist(personne);
        return personne;
    }

    // تحديث شخص
    @TransactionAttribute(TransactionAttributeType.REQUIRED)
    public Personne updateOne(Personne personne) {
        return em.merge(personne);
    }

    • السطر 4: تجعل التعليقة التوضيحية @Stateless من الفئة [Dao] كائن EJB بدون حالة. تجعل التعليقة التوضيحية @Stateful من الفئة كائن EJB مع حالة. يحتوي EJB ذو الحالة على حقول خاصة يجب الحفاظ على قيمتها بمرور الوقت. ومن الأمثلة الكلاسيكية على ذلك فئة تحتوي على معلومات مرتبطة بمستخدم الويب لتطبيق ما. ترتبط مثيل هذه الفئة بمستخدم محدد، وعندما ينتهي مؤشر ترابط تنفيذ طلب هذا المستخدم، يجب الاحتفاظ بالمثيل ليكون متاحًا عند الطلب التالي من نفس العميل. لا يحتوي Ejb @Stateless على حالة. إذا استخدمنا نفس المثال، عند انتهاء مؤشر ترابط تنفيذ طلب مستخدم، سينضم Ejb @Stateless إلى مجموعة من Ejb @Stateless ويصبح متاحًا لمؤشر ترابط تنفيذ طلب مستخدم آخر.
    • بالنسبة للمطور، فإن مفهوم Ejb3 @Stateless قريب من مفهوم singleton في Spring. وسيستخدمه في الحالات نفسها.
    • السطر 7: التعليق التوضيحي @PersistenceContext هو نفسه الموجود في إصدار Spring للطبقة [dao]. وهي تشير إلى الحقل الذي سيستقبل EntityManager الذي سيسمح للطبقة [dao] بمعالجة سياق الاستمرارية.
    • السطر 11: تُستخدم التعليقة التوضيحية @TransactionAttribute المطبقة على إحدى الطرق لتكوين المعاملة التي ستُنفذ فيها الطريقة. فيما يلي بعض القيم المحتملة لهذه التعليقة التوضيحية:
      • TransactionAttributeType.REQUIRED: يجب أن يتم تنفيذ الطريقة في معاملة. إذا كانت هناك معاملة قد بدأت بالفعل، فإن عمليات الاستمرارية الخاصة بالطريقة تتم ضمنها. وإلا، يتم إنشاء معاملة وبدءها.
      • TransactionAttributeType.REQUIRES_NEW: يجب أن يتم تنفيذ الطريقة في معاملة جديدة. يتم إنشاء هذه المعاملة وبدءها.
      • TransactionAttributeType.MANDATORY: يجب أن يتم تنفيذ هذه الطريقة ضمن معاملة قائمة. إذا لم تكن هناك معاملة قائمة، يتم إلقاء استثناء.
      • TransactionAttributeType.NEVER: لا يتم تنفيذ الطريقة أبدًا في معاملة.
      • ...

    دعونا نلقي نظرة على ملفَي التكوين:

    persistence.xml

    
<persistence xmlns="http://java.sun.com/xml/ns/persistence" xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xsi:schemaLocation="http://java.sun.com/xml/ns/persistence
    http://java.sun.com/xml/ns/persistence/persistence_1_0.xsd" version="1.0">

        <!-- المزود JPA هو Hibernate -->

        <!-- تدير بيئة Java DataSource JTA -->
        <jta-data-source>java:/datasource</jta-data-source>

        <properties>
            <!-- البحث عن كيانات الطبقة JBA -->
            <property name="hibernate.archive.autodetection" value="class, hbm" />

            <!-- سجلات SQL Hibernate
                <property name="hibernate.show_sql" value="true"/>
                <property name="hibernate.format_sql" value="true"/>
                <property name="use_sql_comments" value="true"/>
            -->

            <!-- نوع SGBD المدار -->
            <property name="hibernate.dialect" value="org.hibernate.dialect.MySQLInnoDBDialect" />

            <!-- إعادة إنشاء جميع الجداول (drop+create) عند نشر وحدة الاستمرارية -->
            <property name="hibernate.hbm2ddl.auto" value="create" />

        </properties>
    </persistence-unit>

</persistence>

    يشبه هذا الملف تلك التي سبق أن صادفناها في دراسة كيانات JPA. وهو يقوم بتكوين طبقة Hibernate JPA. فيما يلي المستجدات:

    • السطر 5: وحدة الاستمرارية jpa لا تحتوي على السمة transaction-type التي كانت موجودة دائمًا حتى الآن: 
    
<persistence-unit name="jpa" transaction-type="RESOURCE_LOCAL" />

    في حالة عدم وجود قيمة، يكون للسمة transaction-type القيمة الافتراضية "JTA" (لـ Java Transaction Api) التي تشير إلى أن مدير المعاملات يتم توفيره بواسطة حاوية Ejb3. يمكن لمدير "JTA" القيام بأكثر مما يقوم به مدير "RESOURCE_LOCAL": يمكنه إدارة المعاملات التي تغطي عدة اتصالات. باستخدام JTA، يمكن فتح معاملة t1 على اتصال c1 على SGBD 1، معاملة t2 على اتصال c2 مع SGBD 2، ويمكن اعتبار (t1,t2) كمعاملة واحدة إما تنجح فيها جميع العمليات (commit) أو لا تنجح أي منها (rollback).

    هنا، نعمل مع مدير JTA لحاوية Jboss Ejb3.

    3.2.8. تغيير SGBD

    لتغيير SGBD، ما عليك سوى استبدال محتوى المجلد [META-INF] [2] بمحتوى المجلد SGBD الموجود في المجلد [conf] [1]. لنأخذ مثالاً على خادم SQL:

<persistence xmlns="http://java.sun.com/xml/ns/persistence" xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xsi:schemaLocation="http://java.sun.com/xml/ns/persistence
    http://java.sun.com/xml/ns/persistence/persistence_1_0.xsd" version="1.0">

    <persistence-unit name="jpa">

        <!-- المزود JPA هو Hibernate -->
        <provider>org.hibernate.ejb.HibernatePersistence</provider>

        <!-- تدير بيئة Java DataSource JTA -->
        <jta-data-source>java:/datasource</jta-data-source>

        <properties>
            <!-- البحث عن كيانات الطبقة JBA -->
            <property name="hibernate.archive.autodetection" value="class, hbm" />

            <!-- سجلات SQL Hibernate
                <property name="hibernate.show_sql" value="true"/>
                <property name="hibernate.format_sql" value="true"/>
                <property name="use_sql_comments" value="true"/>
            -->

            <!-- نوع SGBD المدار -->
            <property name="hibernate.dialect" value="org.hibernate.dialect.SQLServerDialect" />

            <!-- إعادة إنشاء جميع الجداول (drop+create) عند نشر وحدة الاستمرارية -->
            <property name="hibernate.hbm2ddl.auto" value="create" />

        </properties>
    </persistence-unit>

</persistence>

// init
    @SuppressWarnings("unchecked")
    public void init() throws ServletException {
        try {
            // يتم استرداد معلمات التهيئة الخاصة بـ servlet
            ServletConfig config = getServletConfig();
            // معالجة معلمات التهيئة الأخرى
            String valeur = null;
            for (int i = 0; i < paramètres.length; i++) {
                // قيمة المعلمة
                valeur = config.getInitParameter(paramètres[i]);
                // هل المعلمة موجودة؟
                if (valeur == null) {
                    // تسجيل الخطأ
                    erreursInitialisation.add("Le paramètre [" + paramètres[i] + "] n'a pas été initialisé");
                } else {
                    // يتم حفظ قيمة المعلمة
                    params.put(paramètres[i], valeur);
                }
            }
            // عنوان URL للطريقة [erreurs] له معالجة خاصة
            urlErreurs = config.getInitParameter("urlErreurs");
            if (urlErreurs == null)
                throw new ServletException("Le paramètre [urlErreurs] n'a pas été initialisé");
            // تكوين التطبيق
            // يتم تشغيل الحاوية EJB3 JBoss
            // يتم استخدام ملفات التكوين ejb3-interceptors-aop.xml و embedded-jboss-beans.xml
            EJB3StandaloneBootstrap.boot(null);

            // إنشاء حبوب خاصة بالتطبيق
            EJB3StandaloneBootstrap.deployXmlResource("META-INF/jboss-config.xml");

            // يتم نشر جميع ملفات EJB الموجودة في مسار فئات التطبيق
            //EJB3StandaloneBootstrap.scanClasspath("WEB-INF/classes".replace("/", File.separator));
            EJB3StandaloneBootstrap.scanClasspath();

            // يتم تهيئة السياق JNDI. يتم استخدام الملف jndi.properties
            InitialContext initialContext = new InitialContext();

            // إنشاء مثيل لطبقة الخدمة
            service = (IService) initialContext.lookup("Service/local");
            // يتم إفراغ قاعدة البيانات
            clean();
            // يتم ملؤها
            fill();
        } catch (Exception e) {
            throw new ServletException(e);
        }
    }
